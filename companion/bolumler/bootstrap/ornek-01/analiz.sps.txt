SET DECIMAL=DOT.
GET DATA /TYPE=TXT /FILE='veri.csv' /ENCODING='UTF8'
 /ARRANGEMENT=DELIMITED /DELCASE=LINE /DELIMITERS="," /FIRSTCASE=2
 /VARIABLES=id F8.0 deger F24.16.
DATASET NAME BootRaw.
WEIGHT OFF.
FILTER OFF.
SPLIT FILE OFF.
DATASET COPY BootSummary.
DATASET ACTIVATE BootSummary.
AGGREGATE OUTFILE=* /BREAK= /n=N(deger) /ortalama=MEAN(deger) /orneklem_sd=SD(deger).
COMPUTE ampirik_varyans=orneklem_sd**2*(n-1)/n.
COMPUTE kuramsal_boot_se=SQRT(ampirik_varyans/n).
COMPUTE anahtar=1.
SORT CASES BY anahtar.
DATASET ACTIVATE BootRaw.
DATASET COPY BT1.
DATASET ACTIVATE BT1.
RENAME VARIABLES (id=ind1) (deger=deger1).
SORT CASES BY ind1.
DATASET ACTIVATE BootRaw.
DATASET COPY BT2.
DATASET ACTIVATE BT2.
RENAME VARIABLES (id=ind2) (deger=deger2).
SORT CASES BY ind2.
DATASET ACTIVATE BootRaw.
DATASET COPY BT3.
DATASET ACTIVATE BT3.
RENAME VARIABLES (id=ind3) (deger=deger3).
SORT CASES BY ind3.
DATASET ACTIVATE BootRaw.
DATASET COPY BT4.
DATASET ACTIVATE BT4.
RENAME VARIABLES (id=ind4) (deger=deger4).
SORT CASES BY ind4.
DATASET ACTIVATE BootRaw.
DATASET COPY BT5.
DATASET ACTIVATE BT5.
RENAME VARIABLES (id=ind5) (deger=deger5).
SORT CASES BY ind5.
GET DATA /TYPE=TXT /FILE='bootstrap-plan.csv' /ENCODING='UTF8'
 /ARRANGEMENT=DELIMITED /DELCASE=LINE /DELIMITERS="," /FIRSTCASE=2
 /VARIABLES=sira F8.0 ind1 F8.0 ind2 F8.0 ind3 F8.0 ind4 F8.0 ind5 F8.0.
DATASET NAME BootPlan.
SORT CASES BY ind1.
MATCH FILES /FILE=* /TABLE=BT1 /BY ind1.
EXECUTE.
SORT CASES BY ind2.
MATCH FILES /FILE=* /TABLE=BT2 /BY ind2.
EXECUTE.
SORT CASES BY ind3.
MATCH FILES /FILE=* /TABLE=BT3 /BY ind3.
EXECUTE.
SORT CASES BY ind4.
MATCH FILES /FILE=* /TABLE=BT4 /BY ind4.
EXECUTE.
SORT CASES BY ind5.
MATCH FILES /FILE=* /TABLE=BT5 /BY ind5.
EXECUTE.
COMPUTE boot_ort=(deger1+deger2+deger3+deger4+deger5)/5.
AGGREGATE OUTFILE=* MODE=ADDVARIABLES /BREAK= /toplam=N(boot_ort).
SORT CASES BY boot_ort.
COMPUTE konum=$CASENUM.
COMPUTE h_alt=1+(toplam-1)*0.025.
COMPUTE h_ust=1+(toplam-1)*0.975.
COMPUTE alt_katki=0.
COMPUTE ust_katki=0.
IF (konum=TRUNC(h_alt)) alt_katki=(1-h_alt+TRUNC(h_alt))*boot_ort.
IF (konum=TRUNC(h_alt)+1) alt_katki=(h_alt-TRUNC(h_alt))*boot_ort.
IF (konum=TRUNC(h_ust)) ust_katki=(1-h_ust+TRUNC(h_ust))*boot_ort.
IF (konum=TRUNC(h_ust)+1) ust_katki=(h_ust-TRUNC(h_ust))*boot_ort.
COMPUTE olasilik=1/toplam.
GRAPH /HISTOGRAM=boot_ort.
AGGREGATE OUTFILE=* /BREAK=
 /yeniden_orneklem=N(boot_ort) /boot_merkez=MEAN(boot_ort)
 /boot_sd_b1=SD(boot_ort) /boot_min=MIN(boot_ort) /boot_max=MAX(boot_ort)
 /alt025=SUM(alt_katki) /ust975=SUM(ust_katki) /olasilik_toplami=SUM(olasilik).
COMPUTE boot_varyans=boot_sd_b1**2*(yeniden_orneklem-1)/yeniden_orneklem.
COMPUTE boot_se=SQRT(boot_varyans).
COMPUTE anahtar=1.
SORT CASES BY anahtar.
MATCH FILES /FILE=* /TABLE=BootSummary /BY anahtar.
COMPUTE boot_yanlilik=boot_merkez-ortalama.
FORMATS n yeniden_orneklem (F8.0)
 ortalama ampirik_varyans orneklem_sd kuramsal_boot_se boot_merkez boot_varyans
 boot_se boot_min boot_max alt025 ust975 boot_yanlilik olasilik_toplami (F18.10).
LIST n ortalama ampirik_varyans orneklem_sd kuramsal_boot_se yeniden_orneklem
 boot_merkez boot_varyans boot_se boot_min boot_max alt025 ust975 boot_yanlilik olasilik_toplami.

 /ARRANGEMENT=DELIMITED /DELCASE=LINE /DELIMITERS="," /FIRSTCASE=2
 /VARIABLES=sira F8.0 ind1 F8.0 ind2 F8.0 ind3 F8.0 ind4 F8.0 ind5 F8.0 ind6 F8.0.
DATASET NAME PermPlan.
SORT CASES BY ind1.

SORT CASES BY ind2.
MATCH FILES /FILE=* /TABLE=PT2 /BY ind2.
