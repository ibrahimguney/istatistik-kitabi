GET DATA /TYPE=TXT /FILE='veri.csv' /ENCODING='UTF8'
 /ARRANGEMENT=DELIMITED /DELCASE=LINE /DELIMITERS="," /FIRSTCASE=2
 /VARIABLES=
 M1 F8.2
 M2 F8.2
 M3 F8.2
 M4 F8.2
 M5 F8.2.
EXECUTE.
RECODE M4 (1=5)(2=4)(3=3)(4=2)(5=1) INTO M4_T.
RELIABILITY /VARIABLES=M1 M2 M3 M4_T M5 /SCALE('Bes madde') ALL
 /MODEL=ALPHA /STATISTICS=DESCRIPTIVE SCALE CORR /SUMMARY=TOTAL.
