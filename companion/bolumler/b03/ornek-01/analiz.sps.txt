GET DATA /TYPE=TXT /FILE='veri.csv' /ENCODING='UTF8'
 /ARRANGEMENT=DELIMITED /DELCASE=LINE /DELIMITERS="," /QUALIFIER='"'
 /FIRSTCASE=2 /VARIABLES=saat F16.0.
DATASET NAME B03Ornek.
FILTER OFF.
USE ALL.
WEIGHT OFF.
SPLIT FILE OFF.
VARIABLE LABELS saat 'Haftalik calisma suresi (saat)'.
VARIABLE LEVEL saat (SCALE).
EXECUTE.
DISPLAY DICTIONARY.
DESCRIPTIVES VARIABLES=saat
 /STATISTICS=MEAN STDDEV VARIANCE MIN MAX SUM RANGE.
FREQUENCIES VARIABLES=saat /STATISTICS=MEDIAN /ORDER=ANALYSIS.
SORT CASES BY saat (A).
COMPUTE sira=$CASENUM.
AGGREGATE /OUTFILE=* MODE=ADDVARIABLES /BREAK= /hacim=N.
COMPUTE konum1=1+(hacim-1)*.25.
COMPUTE konum3=1+(hacim-1)*.75.
COMPUTE q1_d=$SYSMIS.
COMPUTE q1_u=$SYSMIS.
COMPUTE q3_d=$SYSMIS.
COMPUTE q3_u=$SYSMIS.
IF (sira=TRUNC(konum1)) q1_d=saat.
IF (sira=MIN(TRUNC(konum1)+1,hacim)) q1_u=saat.
IF (sira=TRUNC(konum3)) q3_d=saat.
IF (sira=MIN(TRUNC(konum3)+1,hacim)) q3_u=saat.
AGGREGATE /OUTFILE=* MODE=ADDVARIABLES /BREAK=
 /q1_alt=MAX(q1_d) /q1_ust=MAX(q1_u)
 /q3_alt=MAX(q3_d) /q3_ust=MAX(q3_u).
COMPUTE q1=q1_alt+(konum1-TRUNC(konum1))*(q1_ust-q1_alt).
COMPUTE q3=q3_alt+(konum3-TRUNC(konum3))*(q3_ust-q3_alt).
COMPUTE iqr=q3-q1.
COMPUTE alt_sinir=q1-1.5*iqr.
COMPUTE ust_sinir=q3+1.5*iqr.
COMPUTE aykiri=(saat<alt_sinir OR saat>ust_sinir).
FORMATS q1 q3 iqr alt_sinir ust_sinir (F10.4).
EXECUTE.
TEMPORARY.
SELECT IF (sira=1).
LIST VARIABLES=q1 q3 iqr alt_sinir ust_sinir.
FREQUENCIES VARIABLES=aykiri.
LIST VARIABLES=sira saat aykiri.
GRAPH /HISTOGRAM=saat.
EXAMINE VARIABLES=saat /PLOT=BOXPLOT /STATISTICS=NONE /NOTOTAL.
TEMPORARY.
SELECT IF (saat<>13).
FREQUENCIES VARIABLES=saat /STATISTICS=MEAN MEDIAN /ORDER=ANALYSIS.
