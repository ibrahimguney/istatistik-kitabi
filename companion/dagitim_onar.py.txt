import argparse
import hashlib
import json
from pathlib import Path, PurePosixPath


PAKETLER = [f'b{number:02}' for number in range(1, 15)] + ['bootstrap', 'coklu-regresyon', 'anova']


def ozet(data):
    return hashlib.sha256(data).hexdigest()


def onarim_plani(kok):
    islemler = []
    for kod in PAKETLER:
        dizin = kok / kod
        belge = json.loads((dizin / 'MANIFEST.json').read_text(encoding='utf-8'))
        if belge.get('algorithm') != 'SHA-256' or not belge.get('files'):
            raise ValueError('Gecerli, bos olmayan manifest gerekli: ' + kod)
        gorulen = set()
        for kaynak_ad, beklenen in belge['files'].items():
            yol = PurePosixPath(kaynak_ad.replace('\\', '/'))
            if (yol.is_absolute() or '..' in yol.parts or ':' in str(yol)
                    or str(yol) == '.' or str(yol) in gorulen):
                raise ValueError('Gecersiz manifest yolu: ' + kaynak_ad)
            gorulen.add(str(yol))
            hedef = dizin / str(yol)
            if any(path.is_symlink() for path in (hedef, *hedef.parents)):
                raise ValueError('Sembolik baglanti reddedildi.')
            once = hedef.read_bytes() if hedef.is_file() else None
            if once is not None and ozet(once) == beklenen:
                continue
            if once is None and not kaynak_ad.endswith('.sps'):
                raise ValueError('Metin yedegiyle onarilamayan eksik dosya: ' + str(hedef))
            kaynak = hedef if once is not None else Path(str(hedef) + '.txt')
            data = kaynak.read_bytes()
            data.decode('utf-8')
            adaylar = [data, data + b'\n',
                       data.replace(b'\r\n', b'\n').replace(b'\n', b'\r\n'),
                       (data.replace(b'\r\n', b'\n').rstrip(b'\n') + b'\n').replace(b'\n', b'\r\n')]
            sonra = next((aday for aday in adaylar if ozet(aday) == beklenen), None)
            if sonra is None:
                raise ValueError('Icerik farki var; otomatik onarim durdu: ' + str(hedef))
            islemler.append((hedef, once, sonra))
    return islemler


def main():
    parser = argparse.ArgumentParser(description='Mevcut manifesti koruyarak eksik SPSS ve satir sonlarini onarir.')
    parser.add_argument('--uygula', action='store_true')
    secenek = parser.parse_args()
    islemler = onarim_plani(Path(__file__).parent / 'bolumler')
    for hedef, once, sonra in islemler:
        print(('GERI YUKLE ' if once is None else 'SATIR SONU ') + str(hedef))
    print('Planlanan dosya:', len(islemler))
    if not secenek.uygula:
        print('Dosyalara yazilmadi. Uygulamak icin --uygula kullanin.')
        return
    for hedef, once, sonra in islemler:
        if (hedef.read_bytes() if hedef.is_file() else None) != once:
            raise ValueError('On incelemeden sonra dosya degismis: ' + str(hedef))
    for hedef, once, sonra in islemler:
        hedef.write_bytes(sonra)
    print('Onarildi; manifestler ve sayisal referanslar degistirilmedi.')


if __name__ == '__main__':
    main()
