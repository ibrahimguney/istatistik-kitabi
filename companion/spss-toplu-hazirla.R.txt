  hazirla <- function() {
    cat('B01/ornek-01/cozum.R dosyasini secin.\n')
    secilen <- normalizePath(file.choose(), winslash='/', mustWork=TRUE)
    ornek_b01 <- dirname(secilen)
    if (basename(secilen) != 'cozum.R' || basename(ornek_b01) != 'ornek-01' ||
        basename(dirname(ornek_b01)) != 'b01') stop('b01/ornek-01/cozum.R secilmeli.')
    kok <- dirname(dirname(ornek_b01))
    paketler <- names(spss_kodlari)
    eksikler <- character()
    for (kod in paketler) {
      girdiler <- file.path(kok, kod, 'ornek-01', gerekli_csv[[kod]])
      eksikler <- c(eksikler, girdiler[!file.exists(girdiler)])
    }
    if (length(eksikler)) stop(paste('Eksik CSV; hazirlik durduruldu:', paste(eksikler, collapse='\n')))
    sonuc <- tempfile(pattern='spss-toplu-sonuc-', tmpdir=dirname(kok))
    if (!dir.create(sonuc)) stop('Sonuc klasoru olusturulamadi.')
    sonuc <- normalizePath(sonuc, winslash='/', mustWork=TRUE)
    tirnak <- function(metin) paste0("'", gsub("'", "''", metin, fixed=TRUE), "'")
    tum_kod <- character()
    kontrol <- NULL
    for (kod in paketler) {
      kaynak <- file.path(kok, kod, 'ornek-01')
      hedef <- file.path(sonuc, 'girdiler', kod)
      if (!dir.create(hedef, recursive=TRUE)) stop('Girdi klasoru olusturulamadi.')
      dosyalar <- file.path(kaynak, gerekli_csv[[kod]])
      if (!all(file.copy(dosyalar, hedef, overwrite=FALSE))) stop('CSV kopyalanamadi.')
      kaynak_md5 <- unname(tools::md5sum(dosyalar))
      kopya_md5 <- unname(tools::md5sum(file.path(hedef, gerekli_csv[[kod]])))
      if (!identical(kaynak_md5, kopya_md5) || anyNA(kaynak_md5)) {
        stop('Kopyalanan girdiler kaynakla eslesmiyor.')
      }
      write.csv(data.frame(dosya=basename(dosyalar), md5=unname(tools::md5sum(dosyalar))),
                file.path(sonuc, paste0(kod, '-girdi-md5.csv')), row.names=FALSE)
      ozet <- read.csv(file.path(hedef, 'beklenen-sonuclar.csv'), check.names=FALSE)
      if (!identical(names(ozet), c('degisken','olcu','deger')) || nrow(ozet)==0) {
        stop('Referans dosyasi gecersiz: ', kod)
      }
      isim <- paste0('Kontrol_', gsub('-', '_', kod, fixed=TRUE))
      baslangic <- c(paste0('OUTPUT NEW NAME=', isim, '.'),
                     paste0('TITLE ', tirnak(paste(kod, 'SPSS kontrol')), '.'),
                     'NEW FILE.', 'SET DECIMAL=DOT.',
                     paste0('CD ', tirnak(hedef), '.'))
      bitis <- c('EXECUTE.', paste0('OUTPUT SAVE OUTFILE=', tirnak(file.path(sonuc, paste0(kod, '.spv'))), '.'),
                 'OUTPUT EXPORT /CONTENTS EXPORT=ALL',
                 paste0(' /PDF DOCUMENTFILE=', tirnak(file.path(sonuc, paste0(kod, '.pdf'))), '.'))
      blok <- c(baslangic, spss_kodlari[[kod]], bitis, '')
      writeLines(enc2utf8(blok), file.path(sonuc, paste0(kod, '-tek.sps.txt')), useBytes=TRUE)
      tum_kod <- c(tum_kod, blok)
      kontrol <- rbind(kontrol, data.frame(paket=kod, referans_satiri=nrow(ozet),
                         durum='CALISTIRILMADI', karsilastirilan_olcu=NA_integer_,
                         uyusmayan_olculer='', pdf=paste0(kod, '.pdf'), spv=paste0(kod, '.spv')))
    }
    tum_kod <- c(tum_kod, paste0('CD ', tirnak(sonuc), '.'))
    writeLines(enc2utf8(tum_kod), file.path(sonuc, 'spss-toplu.sps'), useBytes=TRUE)
    writeLines(enc2utf8(tum_kod), file.path(sonuc, 'spss-toplu.sps.txt'), useBytes=TRUE)
    write.csv(kontrol, file.path(sonuc, 'kontrol-formu.csv'), row.names=FALSE)
    capture.output(sessionInfo(), file=file.path(sonuc, 'hazirlayan-R-oturumu.txt'))
    writeLines(c('Bu klasor R tarafindan hazirlandi; SPSS henuz calistirilmadi.',
                 'Acik calismalari kaydedin; temiz bir SPSS oturumunda spss-toplu.sps dosyasini acin.',
                 'Run > All ile tam dosyayi calistirin; her paket ayri PDF ve SPV uretir.',
                 'Dosya olusmasi veya betigin sona gelmesi sayisal basari kaniti degildir.',
                 'Hata/Uyari mesajlarini ve tum referanslari paket bazinda inceleyin.',
                 'Negatif test yapilmaz; kontrol-formu.csv insan incelemesi bekler.',
                 'Ayni SPSS betigini yeniden calistirmak bu klasordeki ciktilari ezer.',
                 'Yeni deneme icin R hazirlayicisini yeniden calistirip yeni klasor olusturun.',
                 'Girdiler kopyadir; asil bolum paketlerindeki CSV veya manifest degistirilmez.',
                 'MD5 girdi kayitlari paketlerin SHA-256 manifest denetiminin yerine gecmez.'),
               file.path(sonuc, 'ONCE-OKUYUN.txt'))
    cat('\n17 paket icin SPSS betigi HAZIRLANDI; analizler henuz calismadi.\n')
    cat('SPSS ile acilacak dosya:', file.path(sonuc, 'spss-toplu.sps'), '\n')
    cat('Sonuc klasoru:', sonuc, '\n')
    invisible(sonuc)
  }
