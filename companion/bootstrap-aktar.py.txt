SHA256 = 'a9cc8cb8b8309d39f2c90e0fd606d08cefb790fff10d62c909c74bc9fcae9fe4'

def main():
    parser = argparse.ArgumentParser(description='Bootstrap paketini yeni bir goreli klasore acar.')
    parser.add_argument('--hedef', default='bootstrap-aktarim')
    secenek = parser.parse_args()
    hedef = Path(secenek.hedef)
    if (hedef.is_absolute() or '..' in hedef.parts or hedef == Path('.') or hedef.exists()
            or any(path.is_symlink() for path in (hedef, *hedef.parents))):
        parser.error('Yeni, goreli ve sembolik baglanti icermeyen hedef gerekli.')
    data = base64.b64decode(PAYLOAD)
    if hashlib.sha256(data).hexdigest() != SHA256:
        raise ValueError('Dagitim kopyasi bozulmus.')
    with zipfile.ZipFile(io.BytesIO(data)) as archive:
        names = archive.namelist()
        if len(names) != len(set(names)):
            raise ValueError('Yinelenen arsiv girdisi.')
        for name in names:
            path = PurePosixPath(name)
            if (path.is_absolute() or '..' in path.parts or path.parts[0] != 'bootstrap'
                    or ':' in name or '\\' in name):
                raise ValueError('Gecersiz arsiv yolu.')
        manifest = json.loads(archive.read('bootstrap/MANIFEST.json'))
        if set(names) != {'bootstrap/'+name for name in manifest['files']} | {'bootstrap/MANIFEST.json'}:
            raise ValueError('Manifest dosya listesi farkli.')
        for name, expected in manifest['files'].items():
            if hashlib.sha256(archive.read('bootstrap/'+name)).hexdigest() != expected:
                raise ValueError('Manifest uyusmazligi: '+name)
        hedef.mkdir(parents=True, exist_ok=False)
        archive.extractall(hedef)
    print('Acildi:', hedef / 'bootstrap')
    print('Sonraki kontrol: python', hedef / 'bootstrap/ornek-01/cozum.py', '--check')

if __name__ == '__main__':
    main()
